with movimento_mensal as (
    select
        id_cliente,
        nome_cliente,
        ano_mes,
        valor_liquido,
        qtd_transacoes
    from gold_resumo_cliente_mes
),

com_mes_anterior as (
    select
        id_cliente,
        nome_cliente,
        ano_mes,
        valor_liquido,
        qtd_transacoes,
        lag(valor_liquido) over (
            partition by id_cliente
            order by ano_mes
        )                                           as valor_mes_anterior,
        lag(ano_mes) over (
            partition by id_cliente
            order by ano_mes
        )                                           as mes_anterior
    from movimento_mensal
)

select
    id_cliente,
    nome_cliente,
    mes_anterior,
    ano_mes                                         as mes_atual,
    valor_mes_anterior,
    valor_liquido                                   as valor_mes_atual,
    valor_liquido - valor_mes_anterior              as variacao_absoluta,
    round(
        (valor_liquido - valor_mes_anterior) / valor_mes_anterior * 100,
        2
    )                                               as crescimento_percentual,
    qtd_transacoes
from com_mes_anterior
where valor_mes_anterior is not null
  and valor_mes_anterior > 0
  and (valor_liquido - valor_mes_anterior) / valor_mes_anterior > 0.50
order by crescimento_percentual desc

with evolucao as (
    select
        id_cliente,
        nome_cliente,
        ano_mes,
        valor_liquido,
        qtd_transacoes,
        ticket_medio,

        lag(valor_liquido)  over (partition by id_cliente order by ano_mes)
            as valor_mes_anterior,
        lead(valor_liquido) over (partition by id_cliente order by ano_mes)
            as valor_mes_seguinte,

        row_number() over (partition by id_cliente order by ano_mes)
            as numero_do_mes
    from gold_resumo_cliente_mes
)

select
    id_cliente,
    nome_cliente,
    ano_mes,
    numero_do_mes,
    valor_mes_anterior,
    valor_liquido                       as valor_mes_atual,
    valor_mes_seguinte,
    qtd_transacoes,
    ticket_medio,

    case
        when valor_mes_anterior is null then 'primeiro mes'
        when valor_liquido > valor_mes_anterior then 'crescimento'
        when valor_liquido < valor_mes_anterior then 'retracao'
        else 'estavel'
    end                                 as tendencia,

    case
        when valor_mes_anterior is null OR valor_mes_anterior = 0 then null
        else round(
            (valor_liquido - valor_mes_anterior) / valor_mes_anterior * 100, 2
        )
    end                                 as variacao_percentual

from evolucao
order by id_cliente, ano_mes

with total_por_cliente as (
    select
        id_cliente,
        nome_cliente,
        sum(valor_liquido)      as valor_total,
        sum(qtd_transacoes)     as qtd_transacoes,
        sum(valor_estornado)    as valor_estornado,
        count(distinct ano_mes) as meses_ativos
    from gold_resumo_cliente_mes
    group by id_cliente, nome_cliente
),

classificado as (
    select
        id_cliente,
        nome_cliente,
        valor_total,
        qtd_transacoes,
        valor_estornado,
        meses_ativos,
        round(valor_total / qtd_transacoes, 2)          as ticket_medio,
        ntile(4) over (order by valor_total desc)       as quartil,
        dense_rank() over (order by valor_total desc)   as posicao
    from total_por_cliente
)

select
    id_cliente,
    nome_cliente,
    quartil,
    case quartil
        when 1 then 'Q1 - maior movimentacao'
        when 2 then 'Q2 - movimentacao acima da mediana'
        when 3 then 'Q3 - movimentacao abaixo da mediana'
        when 4 then 'Q4 - menor movimentacao'
    end                             as faixa,
    posicao,
    valor_total,
    qtd_transacoes,
    ticket_medio,
    valor_estornado,
    meses_ativos
from classificado
order by quartil, posicao

with por_cidade as (
    select
        cidade_estabelecimento,
        nome_estabelecimento,
        mcc,
        sum(valor_liquido)                          as valor_liquido,
        count(*)                                    as qtd_transacoes,
        count(distinct id_cliente)                  as qtd_clientes,
        sum(case when estornada then 1 else 0 end)  as qtd_estornadas
    from gold_transacoes_cartao
    group by cidade_estabelecimento, nome_estabelecimento, mcc
),

ranqueado as (
    select
        cidade_estabelecimento,
        nome_estabelecimento,
        mcc,
        valor_liquido,
        qtd_transacoes,
        qtd_clientes,
        qtd_estornadas,
        row_number() over (
            partition by cidade_estabelecimento
            order by valor_liquido desc, nome_estabelecimento
        )                                           as posicao,
        dense_rank() over (
            partition by cidade_estabelecimento
            order by valor_liquido desc
        )                                           as posicao_com_empate,
        sum(valor_liquido) over (
            partition by cidade_estabelecimento
        )                                           as valor_total_cidade
    from por_cidade
)

select
    cidade_estabelecimento,
    posicao,
    posicao_com_empate,
    nome_estabelecimento,
    mcc,
    valor_liquido,
    round(valor_liquido / valor_total_cidade * 100, 2) as participacao_na_cidade,
    qtd_transacoes,
    qtd_clientes,
    qtd_estornadas
from ranqueado
where posicao <= 3
order by cidade_estabelecimento, posicao
